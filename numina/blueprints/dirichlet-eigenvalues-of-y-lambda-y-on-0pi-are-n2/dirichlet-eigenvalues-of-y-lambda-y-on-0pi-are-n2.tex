\section{Dirichlet eigenvalues of $-y'' = \lambda y$ on $[0,\pi]$}

We prove that a real number $\lambda$ admits a nontrivial twice-differentiable
solution of $-y'' = \lambda y$ on some open neighbourhood of $[0,\pi]$ with
$y(0) = y(\pi) = 0$ if and only if $\lambda = n^2$ for some positive natural $n$.
The lemmas below only use bare twice-differentiability (existence of $y'$ and
$y''$ on $J$), so the statement does not need full $C^2$ regularity.

Throughout, ``$y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
with $-y'' = \lambda y$'' is shorthand for:
\begin{itemize}
    \item $J \subseteq \mathbb{R}$ is open and $[0,\pi] \subseteq J$;
    \item for every $x \in J$, $y$ is differentiable at $x$ with derivative
        $y'(x) := \mathrm{deriv}\, y\, x$;
    \item for every $x \in J$, $y'$ is differentiable at $x$ with
        $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$.
\end{itemize}

The argument splits into the easy ``if'' direction (the function $\sin(nx)$
realises the eigenvalue $\lambda = n^2$) and the ``only if'' direction, which is
substantively case-split on the sign of $\lambda$. For $\lambda \le 0$ a
convexity argument forces $y \equiv 0$, contradicting nontriviality. For
$\lambda > 0$ an energy argument forces every solution to take the explicit form
$A\sin(\sqrt{\lambda}x) + B\cos(\sqrt{\lambda}x)$, after which the boundary
conditions and nontriviality force $\sqrt{\lambda} \in \mathbb{N}_{>0}$.

\subsection{Preliminaries: derivatives of $\sin(sx)$, $\cos(sx)$ and their combinations}

\begin{lemma}[Derivative of $x \mapsto \sin(sx)$]
    \label{lem:sin_smul_hasDerivAt}
    \lean{LeanEval.Analysis.ODE.sin_smul_hasDerivAt}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \leanok
    For every $s, x \in \mathbb{R}$, the function $t \mapsto \sin(st)$ has
    derivative $s \cos(sx)$ at $x$ (i.e.\ $\mathrm{HasDerivAt}\,(t \mapsto
    \sin(st))\, (s\cos(sx))\, x$).
\end{lemma}
\begin{proof}
    The linear function $t \mapsto s t$ has derivative $s$ at $x$ (\texttt{hasDerivAt\_id}
    composed with \texttt{HasDerivAt.const\_mul}, or
    \texttt{hasDerivAt\_mul\_const}). By the chain rule for $\sin$
    (\texttt{HasDerivAt.sin} applied to \texttt{Real.hasDerivAt\_sin}),
    $t \mapsto \sin(st)$ has derivative $\cos(sx) \cdot s = s\cos(sx)$ at $x$.
\end{proof}

\begin{lemma}[Derivative of $x \mapsto \cos(sx)$]
    \label{lem:cos_smul_hasDerivAt}
    \lean{LeanEval.Analysis.ODE.cos_smul_hasDerivAt}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \leanok
    For every $s, x \in \mathbb{R}$, the function $t \mapsto \cos(st)$ has
    derivative $-s \sin(sx)$ at $x$.
\end{lemma}
\begin{proof}
    As in \cref{lem:sin_smul_hasDerivAt}, $t \mapsto s t$ has derivative $s$ at
    $x$. By the chain rule for $\cos$ (\texttt{HasDerivAt.cos} applied to
    \texttt{Real.hasDerivAt\_cos}), $t \mapsto \cos(st)$ has derivative
    $-\sin(sx) \cdot s = -s\sin(sx)$ at $x$.
\end{proof}

\begin{lemma}[First derivative of $A\sin(sx) + B\cos(sx)$]
    \label{lem:sin_cos_combo_first_deriv}
    \uses{lem:sin_smul_hasDerivAt, lem:cos_smul_hasDerivAt}
    For every $A, B, s, x \in \mathbb{R}$, the function
    $g(t) = A \sin(st) + B \cos(st)$ has derivative
    $A s \cos(sx) - B s \sin(sx)$ at $x$.
\end{lemma}
\begin{proof}
    \uses{lem:sin_smul_hasDerivAt, lem:cos_smul_hasDerivAt}
    By \cref{lem:sin_smul_hasDerivAt} and \cref{lem:cos_smul_hasDerivAt},
    multiplied by the constants $A$ and $B$ via \texttt{HasDerivAt.const\_mul},
    the summands have derivatives $A s \cos(sx)$ and $-B s \sin(sx)$ at $x$.
    Sum via \texttt{HasDerivAt.add}.
\end{proof}

\begin{lemma}[Second derivative of $A\sin(sx) + B\cos(sx)$]
    \label{lem:sin_cos_combo_second_deriv}
    \lean{LeanEval.Analysis.ODE.sin_cos_combo_second_deriv}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:sin_smul_hasDerivAt, lem:cos_smul_hasDerivAt, lem:sin_cos_combo_first_deriv}
    \leanok
    For every $A, B, s, x \in \mathbb{R}$, with $g(t) = A \sin(st) + B \cos(st)$,
    \[ \mathrm{deriv}\bigl(t \mapsto A s \cos(st) - B s \sin(st)\bigr)(x)
       = -s^2 \, g(x). \]
\end{lemma}
\begin{proof}
    \uses{lem:sin_smul_hasDerivAt, lem:cos_smul_hasDerivAt}
    By \cref{lem:cos_smul_hasDerivAt} and \cref{lem:sin_smul_hasDerivAt},
    combined with \texttt{HasDerivAt.const\_mul}, \texttt{HasDerivAt.sub}, the
    function $t \mapsto A s \cos(st) - B s \sin(st)$ has derivative
    $-A s^2 \sin(sx) - B s^2 \cos(sx) = -s^2(A\sin(sx) + B\cos(sx)) = -s^2 g(x)$
    at $x$. Conclude with \texttt{HasDerivAt.deriv}.
\end{proof}

\begin{lemma}[Initial values of $A\sin(sx) + B\cos(sx)$]
    \label{lem:sin_cos_combo_initial_values}
    \uses{lem:sin_cos_combo_first_deriv}
    For every $A, B, s \in \mathbb{R}$, the function
    $g(x) = A \sin(sx) + B \cos(sx)$ satisfies $g(0) = B$ and
    $\mathrm{deriv}\, g\, 0 = A s$.
\end{lemma}
\begin{proof}
    \uses{lem:sin_cos_combo_first_deriv}
    From $\sin 0 = 0$ and $\cos 0 = 1$, $g(0) = A \cdot 0 + B \cdot 1 = B$. By
    \cref{lem:sin_cos_combo_first_deriv} at $x = 0$, the derivative of $g$ at
    $0$ equals $A s \cos 0 - B s \sin 0 = A s$. Apply \texttt{HasDerivAt.deriv}.
\end{proof}

\begin{lemma}[$A\sin(sx) + B\cos(sx)$ solves the harmonic oscillator]
    \label{lem:sin_cos_combo_solves_ode}
    \lean{LeanEval.Analysis.ODE.sin_cos_combo_solves_ode}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:sin_cos_combo_first_deriv, lem:sin_cos_combo_second_deriv, lem:sin_cos_combo_initial_values}
    \leanok
    For every $A, B, s \in \mathbb{R}$, the function
    $g(x) = A \sin(sx) + B \cos(sx)$ is twice differentiable on $\mathbb{R}$ with
    $\mathrm{deriv}\, g\, x = A s \cos(sx) - B s \sin(sx)$ and
    $\mathrm{deriv}(\mathrm{deriv}\, g)\, x = -s^2 \, g(x)$ for every
    $x \in \mathbb{R}$. Moreover $g(0) = B$ and $\mathrm{deriv}\, g\, 0 = A s$.
\end{lemma}
\begin{proof}
    \uses{lem:sin_cos_combo_first_deriv, lem:sin_cos_combo_second_deriv, lem:sin_cos_combo_initial_values}
    \Cref{lem:sin_cos_combo_first_deriv} packaged with \texttt{HasDerivAt.deriv}
    gives $\mathrm{deriv}\, g\, x = A s \cos(sx) - B s \sin(sx)$ for every $x$.
    \Cref{lem:sin_cos_combo_second_deriv} gives the second derivative formula.
    \Cref{lem:sin_cos_combo_initial_values} gives the initial values.
\end{proof}

\subsection{Backward direction: $\sin(nx)$ is an eigenfunction}

\begin{lemma}[Nontrivial witness for $\sin(nx)$]
    \label{lem:sin_nx_nontrivial_witness}
    \lean{LeanEval.Analysis.ODE.sin_nx_nontrivial_witness}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \leanok
    For every positive natural number $n$, there exists $x_0 \in (0,\pi)$ with
    $\sin(n \cdot x_0) \ne 0$.
\end{lemma}
\begin{proof}
    Take $x_0 = \pi/(2n)$. From $n \ge 1$ and $\pi > 0$ we get $0 < \pi/(2n)
    \le \pi/2 < \pi$, so $x_0 \in (0,\pi)$. Moreover
    $n \cdot x_0 = n \cdot \pi/(2n) = \pi/2$ (using $n \ge 1$ so $n \neq 0$), so
    $\sin(n x_0) = \sin(\pi/2) = 1 \neq 0$ by \texttt{Real.sin\_pi\_div\_two}.
\end{proof}

\begin{lemma}[$\sin(nx)$ realises the eigenvalue $n^2$]
    \label{lem:sin_eigenfunction}
    \lean{LeanEval.Analysis.ODE.sin_eigenfunction}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:sin_cos_combo_solves_ode, lem:sin_nx_nontrivial_witness}
    \leanok
    For every positive natural number $n$, the function $y(x) = \sin(nx)$ and
    $J = \mathbb{R}$ satisfy the Dirichlet ODE conditions: $y$ is twice
    differentiable on $\mathbb{R}$ with $\mathrm{deriv}\, y'(x) = -n^2 \, y(x)$,
    $y(0) = y(\pi) = 0$, and there exists $x_0 \in (0,\pi)$ with $y(x_0) \neq 0$.
\end{lemma}
\begin{proof}
    \uses{lem:sin_cos_combo_solves_ode, lem:sin_nx_nontrivial_witness}
    Apply \cref{lem:sin_cos_combo_solves_ode} with $A = 1$, $B = 0$, $s = n$
    (as a real number). The conclusion is that $y(x) = \sin(nx)$ is twice
    differentiable on $\mathbb{R}$ with $\mathrm{deriv}\, y'(x) = -n^2 y(x)$, and
    $y(0) = 0$. For $y(\pi) = \sin(n\pi) = 0$, apply \texttt{Real.sin\_nat\_mul\_pi}.
    Nontriviality is \cref{lem:sin_nx_nontrivial_witness}.
\end{proof}

\subsection{Forward direction, case $\lambda \le 0$: convexity of $y^2$}

\begin{lemma}[First derivative of $y^2$]
    \label{lem:y_sq_first_deriv}
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    (in the shorthand sense of the introduction). Then for every $x \in J$, the
    function $f(t) = y(t)^2 = y(t) \cdot y(t)$ has derivative $2 y(x) y'(x)$
    at $x$.
\end{lemma}
\begin{proof}
    By assumption, $y$ has derivative $y'(x)$ at $x \in J$. Apply the product
    rule \texttt{HasDerivAt.mul} to $y \cdot y$: the resulting derivative is
    $y'(x) \cdot y(x) + y(x) \cdot y'(x) = 2 y(x) y'(x)$.
\end{proof}

\begin{lemma}[Second derivative formula for $y^2$]
    \label{lem:y_sq_second_deriv_formula}
    \lean{LeanEval.Analysis.ODE.y_sq_second_deriv_formula}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$. Let
    $h(x) = 2 y(x) y'(x)$. Then for every $x \in J$, $h$ has derivative
    $2 y'(x)^2 - 2 \lambda y(x)^2$ at $x$.
\end{lemma}
\begin{proof}
    Fix $x \in J$. By hypothesis $y$ has derivative $y'(x)$ at $x$ and $y'$
    has derivative $-\lambda y(x)$ at $x$. Apply \texttt{HasDerivAt.mul} to the
    product $t \mapsto y(t) \cdot y'(t)$: the derivative at $x$ equals
    $y'(x) \cdot y'(x) + y(x) \cdot (-\lambda y(x)) = y'(x)^2 - \lambda y(x)^2$.
    Multiplying by $2$ via \texttt{HasDerivAt.const\_mul} gives the claimed
    derivative.
\end{proof}

\begin{lemma}[Second derivative of $y^2$ is nonnegative when $\lambda \le 0$]
    \label{lem:y_sq_second_deriv_nonneg}
    \lean{LeanEval.Analysis.ODE.y_sq_second_deriv_nonneg}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:y_sq_second_deriv_formula}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$, and $\lambda \le 0$.
    Let $h(x) = 2 y(x) y'(x)$. Then for every $x \in J$,
    $\mathrm{deriv}\, h\, x \ge 0$.
\end{lemma}
\begin{proof}
    \uses{lem:y_sq_second_deriv_formula}
    By \cref{lem:y_sq_second_deriv_formula}, $h$ has derivative
    $2 y'(x)^2 - 2 \lambda y(x)^2$ at $x \in J$. Now $y'(x)^2 \ge 0$ and
    $-\lambda y(x)^2 \ge 0$ (since $\lambda \le 0$ and $y(x)^2 \ge 0$), so
    $2 y'(x)^2 - 2 \lambda y(x)^2 \ge 0$. Conclude with \texttt{HasDerivAt.deriv}.
\end{proof}

\begin{lemma}[$y^2$ is differentiable on $J$]
    \label{lem:y_sq_differentiableOn_J}
    \lean{LeanEval.Analysis.ODE.y_sq_differentiableOn_J}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:y_sq_first_deriv}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    (in the shorthand sense of the introduction). Then the function
    $f(x) = y(x)^2$ is differentiable on $J$, and moreover
    $\mathrm{deriv}\, f\, x = 2 y(x) y'(x)$ for every $x \in J$.
\end{lemma}
\begin{proof}
    \uses{lem:y_sq_first_deriv}
    For each $x \in J$, \cref{lem:y_sq_first_deriv} provides a \texttt{HasDerivAt}
    witness for $f$ at $x$ with derivative $2 y(x) y'(x)$. \texttt{HasDerivAt.differentiableAt}
    gives differentiability at $x$, so $f$ is differentiable on $J$, and
    \texttt{HasDerivAt.deriv} gives the derivative formula.
\end{proof}

\begin{lemma}[$\mathrm{deriv}\, y^2$ is differentiable on $J$]
    \label{lem:y_sq_deriv_differentiableOn_J}
    \lean{LeanEval.Analysis.ODE.y_sq_deriv_differentiableOn_J}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:y_sq_differentiableOn_J, lem:y_sq_second_deriv_formula}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$. Let $f(x) = y(x)^2$.
    Then $\mathrm{deriv}\, f$ is differentiable on $J$.
\end{lemma}
\begin{proof}
    \uses{lem:y_sq_differentiableOn_J, lem:y_sq_second_deriv_formula}
    By \cref{lem:y_sq_differentiableOn_J}, $\mathrm{deriv}\, f\, x = 2 y(x) y'(x)$
    for $x \in J$. By \cref{lem:y_sq_second_deriv_formula}, this function (call it
    $h$) has a \texttt{HasDerivAt} witness at every $x \in J$, hence is
    differentiable at every $x \in J$. Since $\mathrm{deriv}\, f$ and $h$ agree on
    $J$ and $J$ is open, $\mathrm{deriv}\, f$ is differentiable on $J$ as well
    (\texttt{Filter.EventuallyEq.differentiableAt} via \texttt{IsOpen.mem\_nhds}).
\end{proof}

\begin{lemma}[{$y^2$ is convex on $[0,\pi]$ when $\lambda \le 0$}]
    \label{lem:y_sq_convex}
    \lean{LeanEval.Analysis.ODE.y_sq_convex}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:y_sq_differentiableOn_J, lem:y_sq_deriv_differentiableOn_J, lem:y_sq_second_deriv_nonneg}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$, and $\lambda \le 0$.
    Then the function $f(x) = y(x)^2$ is convex on $[0,\pi]$.
\end{lemma}
\begin{proof}
    \uses{lem:y_sq_differentiableOn_J, lem:y_sq_deriv_differentiableOn_J, lem:y_sq_second_deriv_nonneg}
    By \cref{lem:y_sq_differentiableOn_J}, $f$ is differentiable on $J$, hence
    continuous and (since $\mathrm{interior}\,[0,\pi] \subseteq J$) differentiable
    on $\mathrm{interior}\,[0,\pi]$. By \cref{lem:y_sq_deriv_differentiableOn_J},
    $\mathrm{deriv}\, f$ is differentiable on $\mathrm{interior}\,[0,\pi]$, and by
    \cref{lem:y_sq_second_deriv_nonneg}, $\mathrm{deriv}(\mathrm{deriv}\, f)\, x
    \ge 0$ on $\mathrm{interior}\,[0,\pi]$. Apply
    \texttt{convexOn\_of\_deriv2\_nonneg} (using $\pi \ge 0$ for $\mathrm{Set.Icc}\, 0\, \pi$).
\end{proof}

\begin{lemma}[No eigenvalue $\le 0$]
    \label{lem:no_eigen_nonpos}
    \lean{LeanEval.Analysis.ODE.no_eigen_nonpos}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:y_sq_convex}
    \leanok
    If $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$ with
    $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$, $y(0) = y(\pi) = 0$, and
    $\lambda \le 0$, then $y(x) = 0$ for every $x \in [0,\pi]$.
\end{lemma}
\begin{proof}
    \uses{lem:y_sq_convex}
    By \cref{lem:y_sq_convex}, $f(x) = y(x)^2$ is convex on $[0,\pi]$. The
    boundary conditions give $f(0) = f(\pi) = 0$. By the maximum principle for
    convex functions on an interval (\texttt{ConvexOn.le\_max\_of\_mem\_Icc}),
    \[ f(z) \le \max(f(0), f(\pi)) = 0 \quad \text{for } z \in [0,\pi]. \]
    Since $f(z) = y(z)^2 \ge 0$, we conclude $f(z) = 0$, hence $y(z) = 0$ for
    every $z \in [0,\pi]$.
\end{proof}

\subsection{Forward direction, case $\lambda > 0$: energy method}

\begin{lemma}[Derivative of $(y')^2$]
    \label{lem:yprime_sq_hasDerivAt}
    \lean{LeanEval.Analysis.ODE.yprime_sq_hasDerivAt}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$. Then for every
    $x \in J$, the function $t \mapsto y'(t)^2 = y'(t) \cdot y'(t)$ has
    derivative $-2 \lambda y(x) y'(x)$ at $x$.
\end{lemma}
\begin{proof}
    Fix $x \in J$. By hypothesis $y'$ has derivative $-\lambda y(x)$ at $x$.
    Apply \texttt{HasDerivAt.mul} to $t \mapsto y'(t) \cdot y'(t)$: the
    derivative at $x$ equals $(-\lambda y(x)) \cdot y'(x) + y'(x) \cdot
    (-\lambda y(x)) = -2 \lambda y(x) y'(x)$.
\end{proof}

\begin{lemma}[Energy has derivative $0$ pointwise on $J$]
    \label{lem:energy_hasDerivAt_zero}
    \lean{LeanEval.Analysis.ODE.energy_hasDerivAt_zero}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:y_sq_first_deriv, lem:yprime_sq_hasDerivAt}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$. Then for every
    $x \in J$, the function
    $E(x) := \lambda \, y(x)^2 + y'(x)^2$ has derivative $0$ at $x$.
\end{lemma}
\begin{proof}
    \uses{lem:y_sq_first_deriv, lem:yprime_sq_hasDerivAt}
    Fix $x \in J$. By \cref{lem:y_sq_first_deriv}, $y^2$ has derivative
    $2 y(x) y'(x)$ at $x$; scaling by $\lambda$ via \texttt{HasDerivAt.const\_mul}
    gives derivative $\lambda \cdot 2 y(x) y'(x) = 2 \lambda y(x) y'(x)$ for
    $\lambda y^2$ at $x$. By \cref{lem:yprime_sq_hasDerivAt}, $(y')^2$ has
    derivative $-2 \lambda y(x) y'(x)$ at $x$. Adding via \texttt{HasDerivAt.add}
    yields derivative $2 \lambda y(x) y'(x) + (-2 \lambda y(x) y'(x)) = 0$ for
    $E$ at $x$.
\end{proof}

\begin{lemma}[Energy conservation]
    \label{lem:energy_const}
    \lean{LeanEval.Analysis.ODE.energy_const}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:energy_hasDerivAt_zero}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$. Let
    $E(x) = \lambda \, y(x)^2 + y'(x)^2$. Then for every $x \in [0,\pi]$,
    $E(x) = E(0)$.
\end{lemma}
\begin{proof}
    \uses{lem:energy_hasDerivAt_zero}
    By \cref{lem:energy_hasDerivAt_zero}, for every $x \in J$ the function $E$
    has derivative $0$ at $x$, so $E$ is differentiable on $J \supseteq [0,\pi]$
    and $\mathrm{derivWithin}\, E\, [0,\pi]\, x = 0$ for every $x \in [0,\pi]$
    (via \texttt{HasDerivAt.derivWithin}, using that $[0,\pi]$ is convex hence
    \texttt{UniqueDiffOn}). Apply \texttt{constant\_of\_derivWithin\_zero} to
    conclude $E(x) = E(0)$ for every $x \in [0,\pi]$.
\end{proof}

\begin{lemma}[Zero initial conditions force zero solution]
    \label{lem:zero_initial_zero_solution}
    \lean{LeanEval.Analysis.ODE.zero_initial_zero_solution}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:energy_const}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$, $\lambda > 0$,
    $y(0) = 0$, and $y'(0) = 0$. Then $y(x) = 0$ and $y'(x) = 0$ for every
    $x \in [0,\pi]$.
\end{lemma}
\begin{proof}
    \uses{lem:energy_const}
    By \cref{lem:energy_const}, $E(x) = E(0) = \lambda \cdot 0 + 0 = 0$ for every
    $x \in [0,\pi]$, i.e.\ $\lambda y(x)^2 + y'(x)^2 = 0$ on $[0,\pi]$. Both
    summands are nonnegative (using $\lambda > 0$ and that any square is
    $\ge 0$), so each is zero. From $\lambda y(x)^2 = 0$ and $\lambda > 0$ we get
    $y(x) = 0$, and from $y'(x)^2 = 0$ we get $y'(x) = 0$.
\end{proof}

\begin{lemma}[$A\sin(sx) + B\cos(sx)$ solves $-y'' = \lambda y$ when $s = \sqrt{\lambda}$]
    \label{lem:sin_cos_combo_solves_ode_lambda}
    \uses{lem:sin_cos_combo_solves_ode}
    Let $\lambda \ge 0$, $s = \sqrt{\lambda}$, and $A, B \in \mathbb{R}$. Then
    the function $g(x) = A \sin(sx) + B \cos(sx)$ is twice differentiable on
    $\mathbb{R}$ with $\mathrm{deriv}(\mathrm{deriv}\, g)\, x = -\lambda g(x)$
    for every $x \in \mathbb{R}$.
\end{lemma}
\begin{proof}
    \uses{lem:sin_cos_combo_solves_ode}
    By \cref{lem:sin_cos_combo_solves_ode}, $g$ is twice differentiable on
    $\mathbb{R}$ with $\mathrm{deriv}(\mathrm{deriv}\, g)\, x = -s^2 g(x)$.
    By \texttt{Real.sq\_sqrt} applied to $\lambda \ge 0$, $s^2 = \sqrt{\lambda}^2
    = \lambda$, so $-s^2 g(x) = -\lambda g(x)$.
\end{proof}

\begin{lemma}[Initial data of $g$ when $A = y'(0)/s$, $B = y(0)$]
    \label{lem:g_match_initial_data}
    \lean{LeanEval.Analysis.ODE.g_match_initial_data}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:sin_cos_combo_initial_values}
    \leanok
    Let $y : \mathbb{R} \to \mathbb{R}$, $s \in \mathbb{R}$ with $s \neq 0$. Set
    $A = y'(0)/s$, $B = y(0)$, and $g(x) = A \sin(sx) + B \cos(sx)$. Then
    $g(0) = y(0)$ and $\mathrm{deriv}\, g\, 0 = y'(0)$ (where $y'(0)$ denotes
    $\mathrm{deriv}\, y\, 0$).
\end{lemma}
\begin{proof}
    \uses{lem:sin_cos_combo_initial_values}
    By \cref{lem:sin_cos_combo_initial_values}, $g(0) = B = y(0)$ and
    $\mathrm{deriv}\, g\, 0 = A s = (y'(0)/s) \cdot s = y'(0)$ (using
    $s \neq 0$).
\end{proof}

\begin{lemma}[Difference of two ODE solutions still solves the ODE]
    \label{lem:diff_of_two_ode_solutions}
    \lean{LeanEval.Analysis.ODE.diff_of_two_ode_solutions}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \leanok
    Suppose $J \subseteq \mathbb{R}$ is open and $y, g : \mathbb{R} \to \mathbb{R}$
    are twice differentiable on $J$ with $\mathrm{deriv}\, y'(x) = -\lambda y(x)$
    and $\mathrm{deriv}\, g'(x) = -\lambda g(x)$ for every $x \in J$. Let
    $z(x) = y(x) - g(x)$. Then $z$ is twice differentiable on $J$ with
    $\mathrm{deriv}\, z'(x) = -\lambda z(x)$ for every $x \in J$, and
    $\mathrm{deriv}\, z\, 0 = \mathrm{deriv}\, y\, 0 - \mathrm{deriv}\, g\, 0$
    (when $0 \in J$).
\end{lemma}
\begin{proof}
    Fix $x \in J$. By hypothesis $y$ has derivative $y'(x)$ and $g$ has
    derivative $g'(x)$ at $x$; \texttt{HasDerivAt.sub} gives $z$ a derivative
    $y'(x) - g'(x)$ at $x$. Similarly $y'$ and $g'$ have derivatives
    $-\lambda y(x)$ and $-\lambda g(x)$ at $x$; subtracting,
    $\mathrm{deriv}\, z'(x) = -\lambda y(x) - (-\lambda g(x)) = -\lambda z(x)$.
    The derivative-at-$0$ formula follows from \texttt{HasDerivAt.deriv}.
\end{proof}

\begin{lemma}[Difference $y - g$ has zero initial data and satisfies the ODE]
    \label{lem:solution_diff_is_zero_initial_solution}
    \lean{LeanEval.Analysis.ODE.solution_diff_is_zero_initial_solution}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:sin_cos_combo_solves_ode_lambda, lem:g_match_initial_data, lem:diff_of_two_ode_solutions}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$ and $\lambda > 0$.
    Let $s = \sqrt{\lambda} > 0$, $A = y'(0)/s$, $B = y(0)$, and
    $g(x) = A \sin(sx) + B \cos(sx)$. Define $z(x) = y(x) - g(x)$. Then $z$ is
    twice differentiable on $J$ with $\mathrm{deriv}(\mathrm{deriv}\, z)\, x
    = -\lambda z(x)$ on $J$, $z(0) = 0$, and $\mathrm{deriv}\, z\, 0 = 0$.
\end{lemma}
\begin{proof}
    \uses{lem:sin_cos_combo_solves_ode_lambda, lem:g_match_initial_data, lem:diff_of_two_ode_solutions}
    By \cref{lem:sin_cos_combo_solves_ode_lambda} (with $\lambda > 0 \ge 0$),
    $g$ is twice differentiable on $\mathbb{R}$ with $\mathrm{deriv}\, g'(x) =
    -\lambda g(x)$, hence on $J$. By \cref{lem:diff_of_two_ode_solutions},
    $z = y - g$ is twice differentiable on $J$ with $\mathrm{deriv}\, z'(x) =
    -\lambda z(x)$ on $J$, and $\mathrm{deriv}\, z\, 0 = y'(0) -
    \mathrm{deriv}\, g\, 0$. By \cref{lem:g_match_initial_data} (with $s > 0$,
    hence $s \neq 0$), $g(0) = y(0)$ and $\mathrm{deriv}\, g\, 0 = y'(0)$, so
    $z(0) = y(0) - g(0) = 0$ and $\mathrm{deriv}\, z\, 0 = 0$.
\end{proof}

\begin{lemma}[Explicit form of solutions, $\lambda > 0$]
    \label{lem:solution_explicit}
    \lean{LeanEval.Analysis.ODE.solution_explicit}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:zero_initial_zero_solution, lem:solution_diff_is_zero_initial_solution}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$ and $\lambda > 0$. Let
    $s = \sqrt{\lambda} > 0$. Then for every $x \in [0,\pi]$,
    \[ y(x) = \frac{y'(0)}{s} \sin(sx) + y(0) \cos(sx). \]
\end{lemma}
\begin{proof}
    \uses{lem:zero_initial_zero_solution, lem:solution_diff_is_zero_initial_solution}
    With $A = y'(0)/s$, $B = y(0)$, and $g(x) = A \sin(sx) + B \cos(sx)$, set
    $z(x) = y(x) - g(x)$. By
    \cref{lem:solution_diff_is_zero_initial_solution}, $z$ is twice
    differentiable on $J$ with $\mathrm{deriv}(\mathrm{deriv}\, z)\, x =
    -\lambda z(x)$, $z(0) = 0$, $\mathrm{deriv}\, z\, 0 = 0$. By
    \cref{lem:zero_initial_zero_solution}, $z(x) = 0$ for every
    $x \in [0,\pi]$, i.e.\ $y(x) = g(x)$.
\end{proof}

\begin{lemma}[Solution form when $y(0) = 0$, $\lambda > 0$]
    \label{lem:solution_form_when_y0_zero}
    \lean{LeanEval.Analysis.ODE.solution_form_when_y0_zero}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:solution_explicit}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$, $y(0) = 0$, and
    $\lambda > 0$. Let $s = \sqrt{\lambda} > 0$. Then for every
    $x \in [0,\pi]$,
    \[ y(x) = \frac{y'(0)}{s} \sin(sx). \]
\end{lemma}
\begin{proof}
    \uses{lem:solution_explicit}
    By \cref{lem:solution_explicit}, $y(x) = (y'(0)/s)\sin(sx) + y(0)\cos(sx)$
    for $x \in [0,\pi]$. Substituting $y(0) = 0$ eliminates the cosine summand.
\end{proof}

\begin{lemma}[$s > 0$ and $\sin(s\pi) = 0$ force $s$ to be a positive natural]
    \label{lem:sin_pi_smul_eq_zero_pos_to_nat}
    \lean{LeanEval.Analysis.ODE.sin_pi_smul_eq_zero_pos_to_nat}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \leanok
    Let $s \in \mathbb{R}$ with $s > 0$ and $\sin(s \cdot \pi) = 0$. Then there
    exists a positive natural number $n$ with $(n : \mathbb{R}) = s$.
\end{lemma}
\begin{proof}
    By \texttt{Real.sin\_eq\_zero\_iff}, $\sin(s\pi) = 0$ gives an integer $k$
    with $(k : \mathbb{R}) \cdot \pi = s \pi$. Since $\pi > 0$, this yields
    $(k : \mathbb{R}) = s$. From $s > 0$ we get $k > 0$, hence $k \ge 1$ as an
    integer. Set $n = k.\mathrm{toNat}$ (a natural number); then $n \ge 1$ and
    $(n : \mathbb{R}) = (k : \mathbb{R}) = s$.
\end{proof}

\begin{lemma}[Nontriviality forces $y'(0) \neq 0$ when $y(0) = 0$]
    \label{lem:yprime0_ne_zero_of_nontrivial}
    \lean{LeanEval.Analysis.ODE.yprime0_ne_zero_of_nontrivial}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:solution_form_when_y0_zero}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$, $\lambda > 0$,
    $y(0) = 0$, and there exists $x_0 \in (0,\pi)$ with $y(x_0) \neq 0$. Then
    $y'(0) \neq 0$.
\end{lemma}
\begin{proof}
    \uses{lem:solution_form_when_y0_zero}
    Let $s = \sqrt{\lambda} > 0$. By \cref{lem:solution_form_when_y0_zero},
    $y(x) = (y'(0)/s) \sin(sx)$ for every $x \in [0,\pi]$. Suppose for
    contradiction $y'(0) = 0$. Then $y(x) = 0$ for every $x \in [0,\pi]$, and
    in particular $y(x_0) = 0$ (since $x_0 \in (0,\pi) \subseteq [0,\pi]$),
    contradicting $y(x_0) \neq 0$.
\end{proof}

\begin{lemma}[Positive eigenvalues are squares of positive naturals]
    \label{lem:pos_eigen_nat_sq}
    \lean{LeanEval.Analysis.ODE.pos_eigen_nat_sq}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:solution_form_when_y0_zero, lem:sin_pi_smul_eq_zero_pos_to_nat, lem:yprime0_ne_zero_of_nontrivial}
    \leanok
    Suppose $y$ is twice differentiable on an open set $J \supseteq [0,\pi]$
    with $\mathrm{deriv}\, y'(x) = -\lambda \, y(x)$ on $J$, $y(0) = y(\pi) = 0$,
    there exists $x_0 \in (0,\pi)$ with $y(x_0) \neq 0$, and $\lambda > 0$. Then
    $\lambda = n^2$ for some positive natural number $n$.
\end{lemma}
\begin{proof}
    \uses{lem:solution_form_when_y0_zero, lem:sin_pi_smul_eq_zero_pos_to_nat, lem:yprime0_ne_zero_of_nontrivial}
    Let $s = \sqrt{\lambda} > 0$. By \cref{lem:solution_form_when_y0_zero},
    $y(\pi) = (y'(0)/s) \sin(s\pi)$, so the boundary condition $y(\pi) = 0$ gives
    $(y'(0)/s) \sin(s\pi) = 0$. By \cref{lem:yprime0_ne_zero_of_nontrivial},
    $y'(0) \neq 0$; combined with $s > 0$ this forces $\sin(s\pi) = 0$. By
    \cref{lem:sin_pi_smul_eq_zero_pos_to_nat} (using $s > 0$) there is a
    positive natural $n$ with $(n : \mathbb{R}) = s$, and squaring yields
    $\lambda = s^2 = n^2$.
\end{proof}

\subsection{Main theorem}

\begin{theorem}[{Dirichlet eigenvalues of $-y'' = \lambda y$ on $[0,\pi]$}]
    \label{thm:dirichlet_eigenvalues_eq_nat_sq}
    \lean{LeanEval.Analysis.ODE.dirichlet_eigenvalues_eq_nat_sq}
    \leanfile{LeanEval/Analysis/ODE/DirichletEigenvalues.lean}
    \uses{lem:sin_eigenfunction, lem:no_eigen_nonpos, lem:pos_eigen_nat_sq}
    \leanok
    A real $\lambda$ admits a nontrivial twice-differentiable Dirichlet
    eigenfunction on $[0,\pi]$ (in the shorthand sense of the introduction) if
    and only if $\lambda = n^2$ for some positive natural number $n$.
\end{theorem}
\begin{proof}
    \uses{lem:sin_eigenfunction, lem:no_eigen_nonpos, lem:pos_eigen_nat_sq}
    The backward implication is \cref{lem:sin_eigenfunction}: given $n \ge 1$
    with $\lambda = n^2$, the witness is $y(x) = \sin(nx)$ on $J = \mathbb{R}$.

    For the forward implication, suppose $y$, $J$ realise $\lambda$ as a
    Dirichlet eigenvalue. Split on the sign of $\lambda$. If $\lambda \le 0$,
    then by \cref{lem:no_eigen_nonpos}, $y(x) = 0$ for every $x \in [0,\pi]$,
    contradicting the existence of $x_0 \in (0,\pi)$ with $y(x_0) \neq 0$. Hence
    $\lambda > 0$, and by \cref{lem:pos_eigen_nat_sq}, $\lambda = n^2$ for some
    positive natural $n$.
\end{proof}
